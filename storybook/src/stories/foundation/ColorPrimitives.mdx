export const themeVariableNames = () => {
  const names = new Set();
  const walk = (rules) => {
    for (const rule of rules) {
      if (rule.cssRules && !rule.selectorText) {
        walk(rule.cssRules);
      } else if (rule.style && /theme-(light|dark)|\.classic|\.suse/.test(rule.selectorText || '')) {
        for (const name of rule.style) {
          if (name.startsWith('--') && !name.startsWith('--color-')) {
            names.add(name);
          }
        }
      }
    }
  };

  for (const sheet of document.styleSheets) {
    try {
      walk(sheet.cssRules);
    } catch (e) {}
  }

  return [...names];
};

export const resolveThemes = (primitives) => {
  const probe = document.createElement('span');

  document.body.appendChild(probe);
  const render = (value) => {
    probe.style.color = '';
    probe.style.color = value;

    return probe.style.color ? getComputedStyle(probe).color : null;
  };
  const byColor = new Map(primitives.map((p) => [render(p.value), p.name]));
  const colorsIn = (value) => {
    if (/^\d+\s*,\s*\d+\s*,\s*\d+(\s*,\s*[\d.]+)?$/.test(value)) {
      return [`rgba(${ value })`];
    }

    return value && CSS.supports('color', value) ? [value] : value.match(/#[0-9a-f]{3,8}\b|(rgba?|hsla?)\([^)]*\)/gi) || [];
  };
  const names = themeVariableNames();
  const original = document.body.className;
  const result = { light: new Set(), dark: new Set(), unpainted: {}, background: {} };

  for (const [mode, classes] of Object.entries(THEMES)) {
    for (const classNames of classes) {
      document.body.className = classNames;
      const style = getComputedStyle(document.body);

      for (const name of names) {
        for (const color of colorsIn(style.getPropertyValue(name).trim())) {
          const primitive = byColor.get(render(color));

          if (primitive && UNPAINTED.has(name)) {
            (result.unpainted[primitive] ||= new Set()).add(name);
          } else if (primitive) {
            result[mode].add(primitive);
          }
        }
      }
      if (classNames === `theme-${ mode }`) {
        result.background[mode] = style.getPropertyValue('--body-bg').trim();
      }
    }
  }
  document.body.className = original;
  probe.remove();

  return result;
};

export const describeColor = (primitive, themes) => {
  const usage = USAGE[primitive.name] || { uses: 0, files: {}, light: false, dark: false };

  return {
    ...primitive,
    uses:      usage.uses,
    files:     Object.entries(usage.files).sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0])),
    light:     usage.light || themes.light.has(primitive.name),
    dark:      usage.dark || themes.dark.has(primitive.name),
    unpainted: [...themes.unpainted[primitive.name] || []].sort(),
    computed:  computedColors.primitives[primitive.name] || [],
    alpha:     /-a\d+$/.test(primitive.name),
  };
};

export const themeLabel = ({ uses, light, dark }) => {
  if (light && dark) {
    return 'Light + dark';
  }
  if (light || dark) {
    return light ? 'Light only' : 'Dark only';
  }

  return uses ? 'Not painted' : 'Unused';
};

export const groupByFamily = (primitives) => {
  const groups = [];

  for (const p of primitives) {
    const family = p.name.match(/^--color-([a-z]+)/)[1];
    const title = ['white', 'black'].includes(family) ? 'white and black' : family;
    let group = groups.find((g) => g.title === title);

    if (!group) {
      group = { title, primitives: [] };
      groups.push(group);
    }
    group.primitives.push(p);
  }

  return groups;
};

export const copyText = async (text) => {
  try {
    await navigator.clipboard.writeText(text);
  } catch (e) {
    const area = document.createElement('textarea');

    area.value = text;
    area.setAttribute('readonly', '');
    area.style.position = 'absolute';
    area.style.left = '-9999px';
    document.body.appendChild(area);
    area.select();
    document.execCommand('copy');
    area.remove();
  }
};

export const useJson = (url, fallback) => {
  const [data, setData] = useState(fallback);

  useEffect(() => {
    fetch(url)
      .then((response) => (response.ok ? response.json() : fallback))
      .then(setData)
      .catch(() => setData(fallback));
  }, [url]);

  return data;
};

export const Computations = ({ entries }) => {
  const groups = [];

  for (const e of entries) {
    const key = `${ e.expression }|${ JSON.stringify(e.inputs) }`;
    let group = groups.find((g) => g.key === key);

    if (!group) {
      group = { key, expression: e.expression, inputs: e.inputs, places: [] };
      groups.push(group);
    }
    group.places.push(`${ e.file.split('/').pop() }${ e.themes ? ` (${ e.themes.join(', ') })` : '' }`);
  }

  return (
    <ul className="ds-list">
      {groups.map((g) => (
        <li key={g.key}>
          <code>{g.expression}</code>
          {Object.entries(g.inputs).map(([input, value]) => (
            <span key={input}> where <code>{input}</code> is <code>{value}</code></span>
          ))}
          <span className="ds-muted"> in {[...new Set(g.places)].join(', ')}</span>
        </li>
      ))}
    </ul>
  );
};

export const storage = {
  get: (key, fallback) => {
    try {
      const value = localStorage.getItem(key);

      return value === null ? fallback : value;
    } catch (e) {
      return fallback;
    }
  },
  set: (key, value) => {
    try {
      localStorage.setItem(key, value);
    } catch (e) {}
  },
  remove: (key) => {
    try {
      localStorage.removeItem(key);
    } catch (e) {}
  },
};

export const shared = (key, fallback, json = true) => {
  const read = () => {
    const raw = key ? storage.get(key, null) : null;

    if (raw === null || !json) {
      return raw ?? fallback;
    }
    try {
      return JSON.parse(raw) ?? fallback;
    } catch (e) {
      return fallback;
    }
  };
  let value = read();
  const listeners = new Set();

  return {
    get: () => value,
    set: (next) => {
      value = typeof next === 'function' ? next(value) : next;
      if (value === null || value === undefined) {
        value = fallback;
        key && storage.remove(key);
      } else if (key) {
        storage.set(key, json ? JSON.stringify(value) : value);
      }
      listeners.forEach((listener) => listener());
    },
    subscribe: (listener) => {
      listeners.add(listener);

      return () => listeners.delete(listener);
    },
  };
};

export const normalizeUsage = (name, entry, index) => {
  const usage = typeof entry === 'string' ? { route: entry } : entry;

  if (!usage?.route) {
    throw new Error(`Usage ${ index + 1 } of ${ shortName(name) } has no route.`);
  }
  let highlight = usage.highlight;

  if (highlight === undefined) {
    highlight = { color: name };
  } else if (typeof highlight === 'string') {
    highlight = { selector: highlight };
  } else if (highlight) {
    highlight = { ...highlight, color: highlight.color === true ? name : highlight.color ? primitiveName(highlight.color) : undefined };
  }

  return {
    label: usage.label || '',
    route: usage.route.startsWith('/') ? usage.route : `/${ usage.route }`,
    theme: usage.theme || 'light',
    steps: usage.steps || [],
    highlight,
  };
};

export const normalizeColor = (key, entry) => {
  const name = primitiveName(key);
  const value = COLOR_TOKENS[name];

  if (!value) {
    throw new Error(`${ name } is not a color primitive.`);
  }
  const spec = Array.isArray(entry) ? { usages: entry } : entry || {};

  if (typeof spec !== 'object') {
    throw new Error(`The spec of ${ shortName(name) } is an object with usages, a note or both.`);
  }

  return {
    name, value, note: spec.note || '', usages: (spec.usages || []).map((usage, i) => normalizeUsage(name, usage, i))
  };
};

export const normalizeGroups = (raw) => {
  const list = Array.isArray(raw) ? raw : raw?.groups || (raw?.colors ? [raw] : null);

  if (!list) {
    throw new Error('Groups are a list, each with a title and its colors.');
  }

  return list.map((group, g) => {
    const entries = Array.isArray(group?.colors) ? group.colors : Object.keys(group?.colors || {});
    const colors = entries.map((entry) => {
      const name = primitiveName(typeof entry === 'string' ? entry : entry?.name || '');

      if (!COLOR_TOKENS[name]) {
        throw new Error(`${ name } in ${ group?.title || `group ${ g + 1 }` } is not a color primitive.`);
      }

      return shortName(name);
    });

    return { title: String(group?.title || ''), colors: [...new Set(colors)] };
  });
};

export const importDedupeSpec = () => {
  const old = storage.get('color-dedupe-spec', null);

  if (old === null) {
    return;
  }
  try {
    const groups = normalizeGroups(parseSpec(old));
    const builtIn = new Set(BUILT_IN_GROUPS.map((g) => JSON.stringify(g)));

    if (storage.get('color-dedupe-groups', null) === null && groups.some((g) => !builtIn.has(JSON.stringify(g)))) {
      const keys = new Set(groups.map(groupKey));

      storage.set('color-dedupe-groups', JSON.stringify([...groups, ...BUILT_IN_GROUPS.filter((g) => !keys.has(groupKey(g)))]));
    }
  } catch (e) {}
  storage.remove('color-dedupe-spec');
};

export const updateGroups = (groups, index, next) => {
  const before = groups[index];
  const list = !next ? groups.filter((g, i) => i !== index) : index < groups.length ? groups.map((g, i) => (i === index ? next : g)) : [...groups, next];

  if (before && next && groupKey(before) !== groupKey(next)) {
    const [from, to] = [groupKey(before), groupKey(next)];

    choicesStore.set((choices) => {
      const moved = choices[from];

      if (!moved || choices[to]) {
        return choices;
      }
      const kept = moved.candidate === 'custom' || next.colors.includes(shortName(moved.candidate || ''));

      return { ...without(choices, from), [to]: { ...moved, candidate: kept ? moved.candidate : null } };
    });
    openStore.set((open) => open.map((k) => (k === from ? to : k)));
  }
  groupsStore.set(list);
};

export const oklab = (value) => {
  const hex = value.match(/^#([0-9a-f]{6})$/i);
  const [r, g, b, alpha = 1] = hex ? hex[1].match(/../g).map((h) => parseInt(h, 16)) : value.match(/\(([^)]+)\)/)[1].split(',').map(Number);
  const [lr, lg, lb] = [r, g, b].map((c) => (c / 255 <= 0.04045 ? c / 255 / 12.92 : ((c / 255 + 0.055) / 1.055) ** 2.4));
  const l = Math.cbrt(0.4122214708 * lr + 0.5363325363 * lg + 0.0514459929 * lb);
  const m = Math.cbrt(0.2119034982 * lr + 0.6806995451 * lg + 0.1073969566 * lb);
  const s = Math.cbrt(0.0883024619 * lr + 0.2817188376 * lg + 0.6299787005 * lb);

  return [0.2104542553 * l + 0.7936177850 * m - 0.0040720468 * s, 1.9779984951 * l - 2.4285922050 * m + 0.4505937099 * s, 0.0259040371 * l + 0.7827717662 * m - 0.8086757660 * s, alpha];
};

export const LAB = Object.fromEntries(PRIMITIVES.map((p) => [p.name, oklab(p.value)]));

export const labDistance = (x, y) => Math.hypot(LAB[x][0] - LAB[y][0], LAB[x][1] - LAB[y][1], LAB[x][2] - LAB[y][2]) + Math.abs(LAB[x][3] - LAB[y][3]);

export const similarTo = (names, count = 6) => {
  const members = names.map(primitiveName);

  if (!members.length) {
    return [];
  }

  return PRIMITIVES
    .filter((p) => !members.includes(p.name))
    .map((p) => ({ name: p.name, value: p.value, distance: Math.min(...members.map((m) => labDistance(m, p.name))) }))
    .sort((a, b) => a.distance - b.distance)
    .slice(0, count);
};

export const NoUsage = ({ name, note }) => {
  const color = describedColors().colors.find((c) => c.name === name);

  if (note) {
    return note;
  }
  if (!color.light && !color.dark) {
    if (color.unpainted.length) {
      return <>Nothing paints it. Only {color.unpainted.map((v, i) => <span key={v}>{i ? ', ' : ''}<code>{v}</code></span>)} {color.unpainted.length === 1 ? 'holds' : 'hold'} it, and no style, template or script in shell or @rancher/components reads {color.unpainted.length === 1 ? 'that variable' : 'those variables'}.</>;
    }

    return color.uses ? (
      <>Nothing paints it. Only Sass variables in {color.files.map(([file], i) => <span key={file}>{i ? ', ' : ''}<code>{file.split('/').pop()}</code></span>)} refer to it, and nothing compiles them into CSS.</>
    ) : 'Nothing references it.';
  }

  return <>Its entry in <code>color-usages.json</code> does not say where Rancher shows it yet.</>;
};

export const AddToGroup = ({ names }) => {
  const [groups] = useGroups();
  const [target, setTarget] = useState('new');
  const [title, setTitle] = useState('');
  const [added, setAdded] = useState(null);
  const shorts = names.map((n) => shortName(primitiveName(n)));
  const index = target === 'new' ? groups.length : Number(target);
  const group = groups[index] || { title: '', colors: [] };
  const colors = [...new Set([...group.colors, ...shorts])];
  const choose = (value) => {
    setTarget(value);
    setTitle(value === 'new' ? '' : groups[Number(value)]?.title || '');
  };
  const add = () => {
    const next = { ...group, title: title.trim(), colors };

    updateGroups(groups, index, next);
    setAdded(next);
  };

  useEffect(() => setAdded(null), [shorts.join()]);

  return (
    <span className="ds-add-to-group">
      <label>
        Add {shorts.length === 1 ? 'it' : 'them'} to
        <select value={target} onChange={(e) => choose(e.target.value)}>
          <option value="new">a new group</option>
          {groups.map((g, i) => <option key={i} value={i}>{groupTitle(g)}</option>)}
        </select>
      </label>
      <label>
        named
        <input value={title} onChange={(e) => setTitle(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && shorts.length && add()} placeholder={groupTitle({ colors })} />
      </label>
      <button type="button" className="ds-copy" onClick={add} disabled={!shorts.length}>Add</button>
      {added && (

        </span>
      )}
    </span>
  );
};

export const joinUrl = (base, route, params = {}) => {
  try {
    const url = new URL(`${ base.replace(/\/+$/, '') }${ route }`);

    Object.entries(params).forEach(([key, value]) => value !== undefined && value !== '' && url.searchParams.set(key, value));

    return url.toString();
  } catch (e) {
    return '';
  }
};

export const UsageCard = ({
  base, usage, color, candidate, onPick, onDetails, viewport, colors, rings, reload, command, header = true
}) => {
  const stage = useRef(null);
  const grab = useRef(null);
  const frame = useRef(null);
  const drag = useRef(null);
  const latest = useRef({});
  const [status, setStatus] = useState({ state: 'waiting' });
  const [nonce, setNonce] = useState(0);
  const [stageWidth, setStageWidth] = useState(0);
  const [mode, setMode] = useState('focus');
  const [free, setFree] = useState(null);
  const [rect, setRect] = useState(null);
  const [interact, setInteract] = useState(false);
  const [vw, vh] = viewport;
  const width = stageWidth;
  const height = width * vh / vw;
  const fitScale = width / vw;
  const src = joinUrl(base, usage.route);
  const colorsKey = JSON.stringify(colors);

  const focusOn = (r) => {
    const s = clamp(Math.min(width / (r.width + 160), height / (r.height + 160)), fitScale, 2);

    return { s, x: width / 2 - (r.x + r.width / 2) * s, y: height / 2 - (r.y + r.height / 2) * s };
  };
  const view = !width ? { s: 1, x: 0, y: 0 } : mode === 'free' && free ? { s: free.s * fitScale, x: free.x * width, y: free.y * width } : mode === 'focus' && rect ? focusOn(rect) : { s: fitScale, x: 0, y: 0 };
  const moveTo = (next) => {
    setFree({ s: next.s / fitScale, x: next.x / width, y: next.y / width });
    setMode('free');
  };
  const zoomAt = (factor, cx = width / 2, cy = height / 2) => {
    const s = clamp(view.s * factor, fitScale / 2, MAX_ZOOM);

    moveTo({ s, x: cx - (cx - view.x) * (s / view.s), y: cy - (cy - view.y) * (s / view.s) });
  };

  latest.current = { colors, rings, view, zoomAt };
  const send = (message) => frame.current?.contentWindow?.postMessage({ source: SOURCE, ...message }, '*');
  const ringWidth = (s) => Math.max(1, 3 / s);

  useEffect(() => {
    const observer = new ResizeObserver(() => setStageWidth(stage.current?.clientWidth || 0));

    observer.observe(stage.current);

    return () => observer.disconnect();
  }, []);

  useEffect(() => {
    const onWheel = (event) => {
      if (event.ctrlKey || event.metaKey) {
        event.preventDefault();
        const box = grab.current.getBoundingClientRect();

        latest.current.zoomAt(Math.exp(-clamp(event.deltaY, -50, 50) * 0.01), event.clientX - box.left, event.clientY - box.top);
      }
    };

    grab.current.addEventListener('wheel', onWheel, { passive: false });

    return () => grab.current?.removeEventListener('wheel', onWheel);
  }, []);

  useEffect(() => {
    const onMessage = (event) => {
      if (!frame.current || event.source !== frame.current.contentWindow || event.data?.source !== SOURCE) {
        return;
      }
      if (event.data.type === 'hello') {
        send({
          type: 'show', theme: usage.theme, steps: usage.steps, highlight: usage.highlight, ringWidth: ringWidth(latest.current.view.s), colors: latest.current.colors
        });
        send({ type: 'rings', shown: latest.current.rings, width: ringWidth(latest.current.view.s) });
      } else if (event.data.type === 'status') {
        setStatus(event.data);
        if (event.data.rect) {
          setRect(event.data.rect);
        }
        if (['done', 'error'].includes(event.data.state) && document.activeElement === frame.current) {
          frame.current.blur();
        }
      } else if (event.data.type === 'rect') {
        setRect(event.data.rect);
      }
    };

    setStatus({ state: 'waiting' });
    setRect(null);
    window.addEventListener('message', onMessage);
    const timeout = setTimeout(() => setStatus((s) => (s.state === 'waiting' ? { state: 'error', message: 'The page never answered. Is the URL a Rancher share built with the color tool, and does it allow framing?' } : s)), 90000);

    return () => {
      window.removeEventListener('message', onMessage);
      clearTimeout(timeout);
    };
  }, [src, nonce, reload, JSON.stringify(usage)]);

  useEffect(() => send({ type: 'colors', colors }), [colorsKey]);
  useEffect(() => send({ type: 'rings', shown: rings, width: ringWidth(view.s) }), [rings, Math.round(view.s * 100)]);
  useEffect(() => {
    if (command?.n) {
      setMode(command.type);
    }
  }, [command?.n]);

  const startDrag = (event) => {
    if (event.button !== 0) {
      return;
    }
    event.currentTarget.setPointerCapture(event.pointerId);
    drag.current = { px: event.clientX, py: event.clientY, start: view };
  };
  const moveDrag = (event) => {
    const d = drag.current;

    if (d) {
      moveTo({ s: d.start.s, x: d.start.x + event.clientX - d.px, y: d.start.y + event.clientY - d.py });
    }
  };
  const endDrag = () => {
    drag.current = null;
  };
  const onKeyDown = (event) => {
    const step = 40;
    const moves = {
      ArrowLeft: [step, 0], ArrowRight: [-step, 0], ArrowUp: [0, step], ArrowDown: [0, -step]
    };

    if (moves[event.key]) {
      event.preventDefault();
      moveTo({ s: view.s, x: view.x + moves[event.key][0], y: view.y + moves[event.key][1] });
    } else if (event.key === '+' || event.key === '=') {
      zoomAt(1.25);
    } else if (event.key === '-') {
      zoomAt(0.8);
    }
  };

  const open = joinUrl(base, usage.route, {
    theme:          usage.theme,
    highlight:      usage.highlight?.selector,
    highlightText:  usage.highlight?.text,
    highlightColor: usage.highlight?.color,
    colors:         Object.keys(colors).length ? JSON.stringify(colors) : undefined,
  });
  const busy = STATUS[status.state];
  const label = `${ shortName(color.name) }: ${ usage.label || usage.route }`;

  return (
    <figure className={`ds-ucard${ candidate === color.name ? ' ds-ucard--picked' : '' }`}>
      {header && <ColorHeader color={color} candidate={candidate} onPick={onPick} onDetails={onDetails} />}
      <div ref={stage} className="ds-stage" style={{ height }}>
        <iframe
          key={`${ src }|${ nonce }|${ reload }`}
          ref={frame}
          src={src}
          title={label}
          loading="lazy"
          style={{ width: vw, height: vh, transform: `translate(${ view.x }px, ${ view.y }px) scale(${ view.s })` }}
          onLoad={() => send({ type: 'ping' })}
        />
        <div
          ref={grab}
          className="ds-stage-grab"
          style={interact ? { pointerEvents: 'none' } : undefined}
          tabIndex={0}
          role="application"
          aria-label={`${ label }. Arrow keys pan, plus and minus zoom.`}
          onPointerDown={startDrag}
          onPointerMove={moveDrag}
          onPointerUp={endDrag}
          onPointerCancel={endDrag}
          onDoubleClick={() => setMode(mode === 'focus' ? 'fit' : 'focus')}
          onKeyDown={onKeyDown}
        />
        {busy && <div className="ds-frame-status" role="status">{busy}...</div>}
        {status.state === 'login' && (
          <div className="ds-frame-status ds-frame-status--problem" role="status">
            <span>Sign in to <a href={joinUrl(base, '/')} target="_blank" rel="noreferrer">this Rancher</a> in another tab, then reload.</span>
          </div>
        )}
      </div>
      <div className="ds-stage-controls">
        <button type="button" className="ds-copy" onClick={() => zoomAt(0.8)} aria-label="Zoom out">-</button>
        <span className="ds-zoom" aria-live="polite">{Math.round(view.s * 100)}%</span>
        <button type="button" className="ds-copy" onClick={() => zoomAt(1.25)} aria-label="Zoom in">+</button>
        <button type="button" className="ds-copy" aria-pressed={mode === 'fit'} onClick={() => setMode('fit')}>Whole page</button>
        <button type="button" className="ds-copy" aria-pressed={mode === 'focus'} disabled={!rect} onClick={() => setMode('focus')}>Highlight</button>
        <button type="button" className="ds-copy" aria-pressed={interact} onClick={() => setInteract(!interact)} title="Let clicks and scrolling reach the page instead of panning">Use page</button>
        <span className="ds-stage-controls-end">
          <button type="button" className="ds-copy" onClick={() => setNonce((n) => n + 1)}>Reload</button>
          <a className="ds-copy" href={open} target="_blank" rel="noreferrer" title="Opens the page with the highlight and the colors, without the steps">Open</a>
        </span>
      </div>
      <figcaption className="ds-ucard-caption">
        <div className="ds-usage-label">{usage.label || usage.route}</div>
        <div className="ds-muted">
          <code>{usage.route}</code>, {usage.theme}
          {status.state === 'done' && status.element && <>, {status.property ? <code>{status.property}</code> : 'the element'} of <code>{status.element}</code></>}
        </div>
        {status.state === 'error' && <div className="ds-usage-error">{status.message}</div>}
      </figcaption>
    </figure>
  );
};

export const choiceOf = (group, choice) => {
  const custom = choice?.custom || '';

  if (choice?.candidate === 'custom') {
    return { candidate: 'custom', custom, shown: custom.trim(), value: custom.trim() };
  }
  const picked = group.colors.find((c) => c.name === choice?.candidate);

  return {
    candidate: picked?.name || null, custom, shown: picked?.name || null, value: picked?.value || null
  };
};

export const escapeMarkdown = (text) => String(text).replace(/[\\`*_[\]<>|&#]/g, (c) => `\\${ c }`);

export const reportMarkdown = (groups, choices) => {
  const rows = groups.map((group) => {
    const { candidate, value } = choiceOf(group, choices[groupKey(group)]);
    const options = group.colors.map((c) => `\`${ c.name }\``).join(', ');
    let choice = 'Not decided';

    if (candidate === 'custom' && value) {
      choice = `A new color, \`${ value.replace(/`/g, '') }\``;
    } else if (candidate) {
      choice = `\`${ candidate }\` (\`${ value }\`)`;
    }

    return `| ${ escapeMarkdown(group.title) } | ${ options } | ${ choice } |`;
  });
  const decided = groups.filter((group) => choiceOf(group, choices[groupKey(group)]).value).length;

  return [
    '## Color dedupe choices',
    '',
    `${ decided } of ${ groups.length } ${ groups.length === 1 ? 'group' : 'groups' } decided. A group's choice replaces all of its options.`,
    '',
    '| Group | Options | Choice |',
    '| --- | --- | --- |',
    ...rows,
    '',
  ].join('\n');
};

export const Dedupe = () => {
  const [groups, builtIn] = useGroups();
  const [choices, setChoices] = useShared(choicesStore);
  const [opened, setOpened] = useShared(openStore);
  const { base } = useRancher();
  const [active, setActive] = useState(null);
  const [rings, setRings] = useState(true);
  const [size, setSize] = useState(() => Number(storage.get('color-dedupe-size', '') || 560));
  const [reload, setReload] = useState(0);
  const [command, setCommand] = useState({ type: 'focus', n: 0 });
  const [details, setDetails] = useState(null);
  const resolved = useMemo(() => groups.map((g) => ({ title: groupTitle(g), colors: g.colors.map(specOf) })), [groups]);
  const choose = (key, choice) => setChoices((current) => ({ ...current, [key]: choice }));
  const toggle = (key) => {
    setOpened((current) => (current.includes(key) ? current.filter((k) => k !== key) : [...current, key]));
    setActive(key);
  };
  const addGroup = () => {
    updateGroups(groups, groups.length, { title: '', colors: [] });
    setOpened((current) => (current.includes('') ? current : [...current, '']));
    revealStore.set({ key: '', focus: true });
  };

  useEffect(() => {
    const onKey = (event) => {
      if (event.target.closest?.('input, textarea, select') || event.metaKey || event.ctrlKey || event.altKey || document.querySelector('dialog[open]')) {
        return;
      }
      const open = resolved.filter((g) => g.colors.length > 1 && opened.includes(groupKey(g)));
      const group = open.find((g) => groupKey(g) === active) || open[0];

      if (!group) {
        return;
      }
      const key = groupKey(group);

      if (event.key === '0' || event.key === 'Escape') {
        choose(key, { ...choices[key], candidate: null });
      } else if (/^[1-9]$/.test(event.key) && group.colors[Number(event.key) - 1]) {
        choose(key, { ...choices[key], candidate: group.colors[Number(event.key) - 1].name });
      }
    };

    window.addEventListener('keydown', onKey);

    return () => window.removeEventListener('keydown', onKey);
  }, [resolved, opened, active, choices]);

  return (
    <div className="ds-dedupe">
      <style>{'.sbdocs-content { max-width: none !important; min-width: 0; } .sbdocs-toc--custom { display: none !important; }'}</style>
      <p>
        Compare primitives that could be one. Each group opens into cards, one for each place a color's spec says
        Rancher shows it: a {VIEWPORT[0]} x {VIEWPORT[1]} Rancher page with the color made visible and ringed, zoomed in
        on the ring. Drag to pan, <kbd>Ctrl</kbd> + scroll or pinch to zoom, and double-click to switch between the ring
        and the whole page. Pick a color to show it in place of the group's others everywhere, and Original
        (or <kbd>0</kbd>) to go back. <kbd>1</kbd>-<kbd>9</kbd> pick the colors of the group you last used. Rename a
        group, add or remove its colors and add the nearest ones from inside it, or tick swatches on the Primitives tab
        and add them from there. A color's name opens its details. Groups and choices are saved in this browser, and
        the report at the bottom lists the choices.
      </p>
      <div className="ds-dedupe-settings">
        <label>
          Card size
          <input
            type="range"
            min="280"
            max="1440"
            step="20"
            value={size}
            onChange={(e) => {
              setSize(Number(e.target.value));
              storage.set('color-dedupe-size', e.target.value);
            }}
          />
          <span className="ds-muted">{size}px</span>
        </label>
      </div>
      <SpecPanel groups={groups} builtIn={builtIn} />
      <datalist id="ds-primitive-names">
        {PRIMITIVES.map((p) => <option key={p.name} value={shortName(p.name)} />)}
      </datalist>
      <div className="ds-dedupe-settings" role="toolbar" aria-label="Every open card">
        <label><input type="checkbox" checked={rings} onChange={(e) => setRings(e.target.checked)} /> Ring</label>
        <button type="button" className="ds-copy" onClick={() => setCommand((c) => ({ type: 'focus', n: c.n + 1 }))}>Zoom all to rings</button>
        <button type="button" className="ds-copy" onClick={() => setCommand((c) => ({ type: 'fit', n: c.n + 1 }))}>Whole pages</button>
        <button type="button" className="ds-copy" onClick={() => setReload((n) => n + 1)}>Reload all</button>
        <button type="button" className="ds-copy" disabled={!opened.length} onClick={() => setOpened([])}>Close all</button>
        <button type="button" className="ds-copy" onClick={addGroup}>New group</button>
      </div>
      {resolved.map((group, i) => {
        const key = groupKey(group);

        return (
          <DedupeGroup
            key={i}
            group={group}
            raw={groups[i]}
            index={i}
            open={opened.includes(key)}
            onToggle={() => toggle(key)}
            onEdit={(next) => updateGroups(groups, i, next)}
            choice={choices[key]}
            onChoose={(choice) => choose(key, choice)}
            onActivate={() => setActive(key)}
            onDetails={(name) => setDetails({ name, names: group.colors.map((c) => c.name) })}
            base={base}
            size={size}
            rings={rings}
            reload={reload}
            command={command}
          />
        );
      })}
      <Report groups={resolved.filter((g) => g.colors.length > 1)} choices={choices} />
      {details && <Details name={details.name} names={details.names} onOpen={(name) => setDetails({ ...details, name })} onClose={() => setDetails(null)} />}
    </div>
  );
};

<TabPanel tab="primitives">

Use these as `var(--color-...)` instead of writing colors out or computing them from other colors. Click a swatch to see the color live in Rancher, ringed where the app paints it, and tick **Group** on swatches to compare them on the Deduplicate tab.

- `--color-<family>-<step>`: the step is (1 - OKLCH lightness) x 1000, rounded, so a higher step is darker. When that step is taken in the family, the color takes the nearest free step that keeps the family in lightness order.
- `--color-<family>-<step>-a<n>`: the same color at n% opacity, shown over a checkerboard, the light theme background and the dark theme background.
- **Uses** counts the references in shell and @rancher/components.
- **Light + dark**, **Light only** and **Dark only** say which themes paint the color, through a theme variable or a component style. A sun marks the light theme and a moon the dark theme, and a color only the dark theme paints gets a dark badge. A dashed badge means nothing paints it: only a theme variable nothing reads, or a Sass variable that never reaches the CSS, refers to it.
- **Computed** marks a color that master computed from another color (`darken()`, `color-mix()` and the like) before it became a primitive. The details say how.
- **Checked** and **Wrong** are your answers in a color's details to whether the ring sits on it, kept in this browser. Under **Group**, a swatch links to the dedupe groups the color is in.

They are defined in `pkg/rancher-components/src/components/design-system/_color-variables.scss`, and `yarn generate:colors` writes their values to `colors.ts`. A color that isn't here yet gets a new primitive there first.
